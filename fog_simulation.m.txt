clc; clear; close all;

% STEP 1: Deploy Nodes
numNodes = 1000;
areaSize = 100;

nodeX = rand(1, numNodes) * areaSize;
nodeY = rand(1, numNodes) * areaSize;

figure;
scatter(nodeX, nodeY, 10, 'filled');
title('Fog Nodes Deployment');
xlabel('X'); ylabel('Y');
grid on;

% STEP 2: Communication Setup
commRange = 8;
adjMatrix = zeros(numNodes);

for i = 1:numNodes
    for j = i+1:numNodes
        dist = sqrt((nodeX(i)-nodeX(j))^2 + (nodeY(i)-nodeY(j))^2);
        
        if dist <= commRange
            adjMatrix(i,j) = 1;
            adjMatrix(j,i) = 1;
        end
    end
end

G = graph(adjMatrix);

figure;
plot(G, 'XData', nodeX, 'YData', nodeY);
title('Communication Graph');

%% STEP 3: K-Means Clustering

k = 10; % number of clusters

[idx, centroids] = kmeans([nodeX' nodeY'], k);

figure;
gscatter(nodeX, nodeY, idx);
hold on;

% Plot centroids
plot(centroids(:,1), centroids(:,2), 'kx', 'MarkerSize', 15, 'LineWidth', 3);

title('K-Means Clustering with Cluster Heads');
xlabel('X'); ylabel('Y');

%% Cluster Head Selection (Closest to centroid)

clusterHeads = zeros(k,1);

for i = 1:k
    clusterNodes = find(idx == i);
    
    distances = sqrt((nodeX(clusterNodes) - centroids(i,1)).^2 + ...
                     (nodeY(clusterNodes) - centroids(i,2)).^2);
    
    [~, minIdx] = min(distances);
    clusterHeads(i) = clusterNodes(minIdx);
end

% Highlight Cluster Heads
scatter(nodeX(clusterHeads), nodeY(clusterHeads), ...
        100, 'r', 'filled');

legend('Clusters', 'Centroids', 'Cluster Heads');

%% LEACH Clustering

p = 0.05; % probability
leach_CH = [];

for i = 1:numNodes
    if rand < p
        leach_CH = [leach_CH i];
    end
end

figure;
scatter(nodeX, nodeY, 10, 'b'); hold on;
scatter(nodeX(leach_CH), nodeY(leach_CH), 100, 'r', 'filled');

title('LEACH - Random Cluster Heads');
legend('Nodes', 'Cluster Heads');


%% Degree-Based Clustering

degrees = sum(adjMatrix, 2);
[~, idx_deg] = sort(degrees, 'descend');

numCH = 10;
deg_CH = idx_deg(1:numCH);

figure;
scatter(nodeX, nodeY, 10, 'b'); hold on;
scatter(nodeX(deg_CH), nodeY(deg_CH), 100, 'g', 'filled');

title('Degree-Based Cluster Heads');
legend('Nodes', 'Cluster Heads');


%% Energy-Based Clustering

energy = rand(numNodes,1) * 100;

[~, idx_energy] = sort(energy, 'descend');
energy_CH = idx_energy(1:10);

figure;
scatter(nodeX, nodeY, 10, 'b'); hold on;
scatter(nodeX(energy_CH), nodeY(energy_CH), 100, 'm', 'filled');

title('Energy-Based Cluster Heads');
legend('Nodes', 'Cluster Heads');



%% COMPARISON VISUALIZATION

figure;

subplot(2,2,1)
gscatter(nodeX, nodeY, idx);
hold on;
scatter(nodeX(clusterHeads), nodeY(clusterHeads), 80, 'r', 'filled');
title('K-Means')

subplot(2,2,2)
scatter(nodeX, nodeY, 10, 'b'); hold on;
scatter(nodeX(leach_CH), nodeY(leach_CH), 80, 'r', 'filled');
title('LEACH')

subplot(2,2,3)
scatter(nodeX, nodeY, 10, 'b'); hold on;
scatter(nodeX(deg_CH), nodeY(deg_CH), 80, 'g', 'filled');
title('Degree-Based')

subplot(2,2,4)
scatter(nodeX, nodeY, 10, 'b'); hold on;
scatter(nodeX(energy_CH), nodeY(energy_CH), 80, 'm', 'filled');
title('Energy-Based')



%% STEP 5: CYBER ATTACK SIMULATION

% Copy original QoS values
throughput_original = degrees * 10;
PDR_original = rand(numNodes,1)*0.2 + 0.8;

throughput = throughput_original;
PDR = PDR_original;

% ---- Attack 1: DoS Attack ----
dos_nodes = randperm(numNodes, round(0.1*numNodes)); % 10% nodes

throughput(dos_nodes) = throughput(dos_nodes) * 0.3;
PDR(dos_nodes) = PDR(dos_nodes) * 0.5;

% ---- Attack 2: Node Failure ----
fail_nodes = randperm(numNodes, round(0.05*numNodes)); % 5%

throughput(fail_nodes) = 0;
PDR(fail_nodes) = 0;

% ---- Attack 3: Cluster Head Attack ----
ch_attack = clusterHeads(1:min(3,length(clusterHeads)));

throughput(ch_attack) = throughput(ch_attack) * 0.1;
PDR(ch_attack) = PDR(ch_attack) * 0.2;



%% QoS Metrics

latency = rand(numNodes,1)*10; % simulated latency

avg_before = [mean(latency), mean(throughput_original), mean(PDR_original)];
avg_after  = [mean(latency), mean(throughput), mean(PDR)];

figure;
bar([avg_before; avg_after]');
set(gca, 'XTickLabel', {'Latency','Throughput','PDR'});
legend('Before Attack','After Attack');

title('QoS Degradation due to Cyberattack');



%% RISK ASSESSMENT

% Impact (QoS degradation)
impact = (throughput_original - throughput) ./ throughput_original;
impact(isnan(impact)) = 0;

% Vulnerability (based on degree + inverse energy)
vulnerability = (degrees / max(degrees)) .* (1 - energy / max(energy));

% Likelihood (based on attack probability)
likelihood = zeros(numNodes,1);
likelihood(dos_nodes) = 0.7;
likelihood(fail_nodes) = 0.9;
likelihood(ch_attack) = 1.0;

% Risk Score
risk = impact .* vulnerability .* likelihood;

% Normalize
risk = risk / max(risk);

% High-risk nodes
highRisk = find(risk > 0.6);


figure;
scatter(nodeX, nodeY, 10, risk, 'filled');
colorbar;
title('Risk Map of Network');

hold on;
scatter(nodeX(highRisk), nodeY(highRisk), 50, 'r', 'filled');
